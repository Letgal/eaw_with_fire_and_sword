on_actions = {
on_startup = {
effect = {
	every_possible_country = {
		limit = {
			is_griffon_race = yes
			NOT = { original_tag = GRW original_tag = BRF }
		}
		# (*) all griffons despise Prywhenians. ~ L.H.
		add_opinion_modifier = { target = GRW modifier = opinion_prywhenians }
		add_opinion_modifier = { target = BRF modifier = opinion_prywhenians }
	}
	every_country = {
		limit = {
			OR = {
				original_tag = GRY
				original_tag = LUS
				original_tag = GRU
				original_tag = BAN
			}
		}
		add_opinion_modifier = { target = GRW modifier = opinion_pry_commies }
	}
}
}

on_government_change = {
effect = {
	every_country = {
		limit = {
			OR = {
				original_tag = GRY
				original_tag = LUS
				original_tag = GRU
				original_tag = BAN
			}
		}
		# (*) Evi valley non-communist nations also despise PLA. ~ L.H.
		if = {
			limit = { has_government = communism }
			remove_opinion_modifier = { target = GRW modifier = opinion_pry_commies }
			else = { add_opinion_modifier = { target = GRW modifier = opinion_pry_commies } }
		}
	}
}
}
}
