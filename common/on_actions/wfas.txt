on_actions = {
on_startup = {
effect = {
	every_possible_country = {
		limit = { is_griffon_race = yes }
		# all griffons despise Prywhenians. ~ L.H.
		add_opinion_modifier = { target = GRW modifier = opinion_prywhenians }
		add_opinion_modifier = { target = BRF modifier = opinion_prywhenians }
	}
	every_country = {
		limit = {
			OR = {
				original_tag = GRY
				original_tag = LUS
				original_tag = GRU
				original_tag = BAN
			}
		}
		add_opinion_modifier = { target = GRW modifier = opinion_pry_commies }
	}

	# GRY
	613 = { set_state_name = wfas_STATE_613 }
	631 = { set_state_name = wfas_STATE_631 }

	# LUS
	591 = { set_state_name = wfas_STATE_591 }
	593 = { set_state_name = wfas_STATE_593 }
	607 = { set_state_name = wfas_STATE_607 }
	608 = { set_state_name = wfas_STATE_608 }
	632 = { set_state_name = wfas_STATE_632 }
	LUS = {
		set_province_name = { id = 8003 name = wfas_VP_8003 }
		set_province_name = { id = 12040 name = wfas_VP_12040 }
		set_province_name = { id = 8026 name = wfas_VP_8026 }
		set_province_name = { id = 11912 name = wfas_VP_11912 }
		set_province_name = { id = 7204 name = wfas_VP_7204 }
		set_province_name = { id = 5071 name = wfas_VP_5071 }
		set_province_name = { id = 4167 name = wfas_VP_4167 }
		set_province_name = { id = 438 name = wfas_VP_438 }
		set_province_name = { id = 10765 name = wfas_VP_10765 }
		set_province_name = { id = 12642 name = wfas_VP_12642 }
	}
}
}

on_government_change = {
effect = {
	every_country = {
		limit = {
			OR = {
				original_tag = GRY
				original_tag = LUS
				original_tag = GRU
				original_tag = BAN
			}
		}
		# Evi valley non-communist nations despise PLA. ~ L.H.
		if = {
			limit = { has_government = communism }
			remove_opinion_modifier = { target = GRW modifier = opinion_pry_commies }
			else = { add_opinion_modifier = { target = GRW modifier = opinion_pry_commies } }
		}
	}
}
}
}
