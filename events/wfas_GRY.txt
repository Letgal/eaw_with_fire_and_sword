		set_country_flag = coronation_costs_decided

	option = {
		name = WFAS_gryphia.6.c
		trigger = {
			has_war = no
			NOT = { has_country_flag = archons_refused_coronation }
		}
		hidden_effect = {
			save_event_target_as = petitioner

					random_list = {
						1 = { GRI = { country_event = { id = WFAS_gryphia.7 days = 3 } } }

					}
				}
			}
		}
		custom_effect_tooltip = petition_to_archons
		custom_effect_tooltip = if_they_accept
		effect_tooltip = {
			add_political_power = -250
			add_timed_idea = { idea = GRY_coronation_costs_3_temp months = 9 }
		}
	}
	option = {
		name = WFAS_gryphia.6.e
		trigger = { has_war = yes }

	}
